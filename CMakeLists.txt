cmake_minimum_required(VERSION 3.20)
project(PulseGrid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(pulsegrid src/main.cpp)
target_include_directories(pulsegrid PRIVATE include)
enable_testing()
add_executable(pulsegrid_tests tests/test_server.cpp)
target_include_directories(pulsegrid_tests PRIVATE include)
add_test(NAME pulsegrid_tests COMMAND pulsegrid_tests)
